        // --- Web Audio API สังเคราะห์เสียงย้อนยุค ---

        const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

        function playTone(freq, type, duration) {

            const osc = audioCtx.createOscillator();

            const gain = audioCtx.createGain();

            osc.type = type;

            osc.frequency.setValueAtTime(freq, audioCtx.currentTime);

            gain.gain.setValueAtTime(0.2, audioCtx.currentTime);

            gain.gain.exponentialRampToValueAtTime(0.00001, audioCtx.currentTime + duration);

            osc.connect(gain);

            gain.connect(audioCtx.destination);

            osc.start();

            osc.stop(audioCtx.currentTime + duration);

        }

        // วาดองค์ประกอบบน Canvas

        function draw() {

            ctx.clearRect(0, 0, canvas.width, canvas.height);

            const currentMap = robot.currentMap || levels[currentLevel].map;


            // 1. วาดตารางเมือง

            for (let r = 0; r < GRID_SIZE; r++) {

                for (let c = 0; c < GRID_SIZE; c++) {

                    ctx.lineWidth = 1;

                    ctx.strokeRect(c * TILE_SIZE, r * TILE_SIZE, TILE_SIZE, TILE_SIZE);


                    // วาดวัตถุตามประเภทพื้นผิว

                    if (currentMap[r][c] === 1) { // ต้นไม้/สิ่งกีดขวาง

                        ctx.fillStyle = "#2ecc71";

                        ctx.fillRect(c * TILE_SIZE + 5, r * TILE_SIZE + 5, TILE_SIZE - 10, TILE_SIZE - 10);

                        ctx.fillStyle = "#27ae60";

                        ctx.font = "20px Arial";

                        ctx.fillText("🌳", c * TILE_SIZE + 20, r * TILE_SIZE + 50);

                    } else if (currentMap[r][c] === 2) { // ตึกที่ยังไฟดับ

                        ctx.fillStyle = "#34495e";

                        ctx.fillRect(c * TILE_SIZE + 8, r * TILE_SIZE + 8, TILE_SIZE - 16, TILE_SIZE - 16);

                        ctx.fillStyle = "white";

                        ctx.font = "18px Arial";

                        ctx.fillText("🏢 ❌", c * TILE_SIZE + 15, r * TILE_SIZE + 48);

                    } else if (currentMap[r][c] === 3) { // ตึกที่เปิดไฟสำเร็จ

                        ctx.fillStyle = "#f1c40f";

                        ctx.fillRect(c * TILE_SIZE + 8, r * TILE_SIZE + 8, TILE_SIZE - 16, TILE_SIZE - 16);

                        ctx.fillStyle = "#2c3e50";

                        ctx.font = "18px Arial";

                        ctx.fillText("🏢 ⚡", c * TILE_SIZE + 15, r * TILE_SIZE + 48);

                    }

                }

            }


            // 2. วาดหุ่นยนต์ผู้เล่น

            ctx.save();

            // ย้ายจุดศูนย์กลางไปที่กล่องที่หุ่นยนต์อยู่

            const rx = robot.x * TILE_SIZE + TILE_SIZE / 2;

            const ry = robot.y * TILE_SIZE + TILE_SIZE / 2;

            ctx.translate(rx, ry);

            // หมุนมุมตามทิศ (0=บน=0องศา, 1=ขวา=90องศา, 2=ล่าง=180องศา, 3=ซ้าย=270องศา)

            ctx.fillStyle = "#1a5276";

            ctx.fillRect(-5, -32, 10, 7); // เสาอากาศ


            ctx.restore();

        }


        // --- ส่วนควบคุมบล็อกคำสั่ง (Pipeline) ---

        function addCommand(cmdType) {

            if (isRunning) return;

            sounds.click();

            program.push(cmdType);

            renderPipeline();

        }


        function removeCommand(index) {

            if (isRunning) return;

            sounds.click();

            program.splice(index, 1);

            renderPipeline();

        }


        function clearPipeline() {

            if (isRunning) return;

            program = [];

            renderPipeline();

            initLevel(currentLevel); // รีเซ็ตตำแหน่งหุ่นยนต์ในด่านปัจจุบันด้วย

        }


        function renderPipeline() {

            const pipelineEl = document.getElementById("pipeline");

            pipelineEl.innerHTML = "";

            

            const translateMap = {

                'FORWARD': { text: '➡ เดินหน้า', class: 'b-move' },

                'TURN_LEFT': { text: '↩ เลี้ยวซ้าย', class: 'b-turn' },

                'TURN_RIGHT': { text: '↪ เลี้ยวขวา', class: 'b-turn' },

                'ACTIVATE': { text: '⚡ ปล่อยไฟฟ้า', class: 'b-action' }

            };


            program.forEach((cmd, idx) => {

                const item = document.createElement("div");

                item.className = `pipeline-item block ${translateMap[cmd].class}`;

                item.innerText = `${idx + 1}. ${translateMap[cmd].text}`;

                

                const delBtn = document.createElement("button");

                delBtn.className = "btn-delete";

                delBtn.innerText = "×";

                delBtn.onclick = (e) => { e.stopPropagation(); removeCommand(idx); };

                

                item.appendChild(delBtn);

                pipelineEl.appendChild(item);

            });

        }


        // --- ตัวแปลภาษาและประมวลผลคำสั่งทีละบรรทัด (Interpreter) ---

        async function runProgram() {

            if (program.length === 0 || isRunning) return;

            isRunning = true;

            document.getElementById("btnRun").disabled = true;

            document.getElementById("btnClear").disabled = true;

            document.getElementById("statusDisplay").innerText = "🤖 รหัสกำลังทำงาน...";

            document.getElementById("statusDisplay").style.color = "#2980b9";


            const pipelineItems = document.getElementById("pipeline").children;


            // ลูปทำงานตามคำสั่งที่เด็กๆ เรียงไว้ทีละขั้น

            for (let i = 0; i < program.length; i++) {

                // ไฮไลต์บล็อกสีทองขณะที่ทำงานอยู่ (เหมือน Scratch เวลาโค้ดรัน)

                if (pipelineItems[i]) pipelineItems[i].style.border = "3px solid gold";

                

                let success = executeCommand(program[i]);

                draw();

                

                if (!success) {

                    sounds.fail();

                    document.getElementById("statusDisplay").innerText = "💥 เกิดข้อผิดพลาดในตรรกะ!";

                    document.getElementById("statusDisplay").style.color = "#c0392b";

                    isRunning = false;

                    document.getElementById("btnRun").disabled = false;

                    document.getElementById("btnClear").disabled = false;

                    return; // หยุดรันโปรแกรมทันทีเมื่อชนกำแพงหรือตกแผนที่

                }


                await new Promise(resolve => setTimeout(resolve, 600)); // หน่วงเวลาให้เด็กๆ ดูทัน

                if (pipelineItems[i]) pipelineItems[i].style.border = "none";

            }


            // หลังวิ่งจบลูปคำสั่ง ตรวจสอบเงื่อนไขการชนะ

            checkWinCondition();

        }


        function executeCommand(cmd) {

            sounds.click();

            switch (cmd) {

                case 'FORWARD':

                    let nx = robot.x;

                    let ny = robot.y;

                    if (robot.dir === 0) ny--;      // เหนือ

                    if (robot.dir === 1) nx++;      // ตะวันออก

                    if (robot.dir === 2) ny++;      // ใต้

                    if (robot.dir === 3) nx--;      // ตะวันตก


                    // ตรวจสอบการชนสิ่งกีดขวางหรือขอบแผนที่

                    if (nx < 0 || nx >= GRID_SIZE || ny < 0 || ny >= GRID_SIZE) return false; 

                    if (robot.currentMap[ny][nx] === 1) return false; // ชนต้นไม้


                    robot.x = nx;

                    robot.y = ny;

                    return true;


                case 'TURN_LEFT':

                    robot.dir = (robot.dir + 3) % 4; // หมุนทวนเข็ม

                    return true;


                case 'TURN_RIGHT':

                    robot.dir = (robot.dir + 1) % 4; // หมุนตามเข็ม

                    return true;


                case 'ACTIVATE':

                    // ปล่อยไฟฟ้าเปิดสวิตช์ตึก ณ จุดที่หุ่นยนต์ยืนอยู่

                    if (robot.currentMap[robot.y][robot.x] === 2) {

                        robot.currentMap[robot.y][robot.x] = 3; // เปลี่ยนสถานะเป็นตึกเปิดไฟแล้ว

                        score += 50;

                        document.getElementById("currentScore").innerText = score;

                        sounds.success();

                    }

                    return true;

            }

            return false;

        }


        function checkWinCondition() {

            isRunning = false;

            document.getElementById("btnRun").disabled = false;

            document.getElementById("btnClear").disabled = false;


            // ตรวจว่าในแผนที่ปัจจุบันยังมีตึกที่ไฟดับ (หมายเลข 2) หลงเหลืออยู่ไหม

            let hasDarkBuilding = false;

            for (let r = 0; r < GRID_SIZE; r++) {

                for (let c = 0; c < GRID_SIZE; c++) {

                    if (robot.currentMap[r][c] === 2) hasDarkBuilding = true;

                }

            }


            if (!hasDarkBuilding) {

                sounds.win();

                document.getElementById("statusDisplay").innerText = "🎉 สำเร็จ! ผ่านด่านแล้ว";

                document.getElementById("statusDisplay").style.color = "#27ae60";

                

                // บันทึกสถิติสูงสุดลงเครื่อง (LocalStorage)

                let high = localStorage.getItem("codeTownHighScore") || 0;

                if (score > high) {

                    localStorage.setItem("codeTownHighScore", score);

                    document.getElementById("highScore").innerText = score;

                }


                setTimeout(() => {

                    currentLevel++;

                    initLevel(currentLevel);

                }, 1500);

            } else {

                sounds.fail();

                document.getElementById("statusDisplay").innerText = "❌ ไฟยังเปิดไม่ครบทุกตึก ลองวางแผนใหม่นะ!";

                document.getElementById("statusDisplay").style.color = "#e67e22";

            }

        }


        // โหลดคะแนนสูงสุดเดิม

        document.getElementById("highScore").innerText = localStorage.getItem("codeTownHighScore") || 0;

        

        // เริ่มต้นเกมแรก

        initLevel(currentLevel);

    </script>
